import axios from "axios";
import { useState } from "react";

const TryOnPage = () => {
  const [userImage, setUserImage] = useState(null);
  const [clothImage, setClothImage] = useState(null);
  const [resultUrl, setResultUrl] = useState("");
  const [loading, setLoading] = useState(false);
  const [errorMessage, setErrorMessage] = useState("");

  const handleFileChange = (e, type) => {
    const file = e.target.files[0];
    if (file) {
      if (type === "user") setUserImage(file);
      if (type === "cloth") setClothImage(file);
    }
  };

  const handleTryOn = async () => {
    try {
      if (!userImage || !clothImage) {
        console.error("Error: Both user image and cloth image are required.");
        setErrorMessage("Both user image and cloth image are required.");
        return;
      }

      const formData = new FormData();
      formData.append("userImage", userImage);
      formData.append("clothImage", clothImage);

      setLoading(true);
      setErrorMessage("");

      console.log("Sending Try-On request...");
      console.log(
        "FormData entries:",
        formData.get("userImage"),
        formData.get("clothImage")
      );

      // ✅ Ensure this request is a POST request
      const response = await axios.post(
        "http://localhost:5050/api/v1/tryon",
        formData, // Sending form data
        {
          headers: {
            "Content-Type": "multipart/form-data",
          },
        }
      );

      console.log("Try-On API Response:", response.data);

      if (response.data && response.data.cropped_target_url) {
        setResultUrl(response.data.cropped_target_url);
        setFinalOutputVisible(true);
      } else {
        setErrorMessage("Try-On result not available.");
      }
    } catch (error) {
      console.error("Error in Try-On:", error);
      setErrorMessage("An error occurred during the Try-On process.");
    } finally {
      setLoading(false);
    }
  };


  return (
    <div>
      <h1>Virtual Try-On</h1>
      <input
        type="file"
        accept="image/*"
        onChange={(e) => handleFileChange(e, "user")}
      />
      <input
        type="file"
        accept="image/*"
        onChange={(e) => handleFileChange(e, "cloth")}
      />
      <button onClick={handleTryOn} disabled={loading}>
        {loading ? "Processing..." : "Try On"}
      </button>

      {errorMessage && <p style={{ color: "red" }}>{errorMessage}</p>}

      {resultUrl && (
        <div>
          <h3>Try-On Result</h3>
          <img src={resultUrl} alt="Try-on Result" style={{ width: "300px" }} />
        </div>
      )}
    </div>
  );
};

export default TryOnPage;
