import axios from "axios";
import { useState } from "react";
import { useLocation } from "react-router-dom";

const TryOnPage = () => {
  const location = useLocation();
  const clothImageUrl = location.state?.clothImage || ""; // Assuming clothImage is a URL

  const [userImage, setUserImage] = useState(null);
  const [resultUrl, setResultUrl] = useState("");
  const [loading, setLoading] = useState(false);
  const [errorMessage, setErrorMessage] = useState("");

  // Handle file selection
  const handleFileChange = (e) => {
    const file = e.target.files[0];
    if (file) {
      setUserImage(file);
    }
  };

  // Upload an image to the backend
  const uploadImage = async (imageFile, fieldName, endpoint) => {
    const formData = new FormData();
    formData.append(fieldName, imageFile);

    try {
      const response = await axios.post(endpoint, formData);
      return response.data.url; // Assuming response contains { url: "uploaded_image_url" }
    } catch (error) {
      console.error(`Error uploading image: ${error}`);
      setErrorMessage("Failed to upload image.");
      return null;
    }
  };

  // Try-On process
  const handleTryOn = async () => {
    if (!userImage || !clothImageUrl) {
      setErrorMessage("Both user image and cloth image are required.");
      return;
    }

    setLoading(true);
    setErrorMessage("");

    try {
      console.log("Uploading user image...");

      // Upload user image
      const userImageUrl = await uploadImage(
        userImage,
        "image", // Correct field name for user image
        "http://localhost:5050/apps/proxy/upload"
      );
      if (!userImageUrl) throw new Error("User image upload failed.");

      console.log("User Image URL:", userImageUrl);
      console.log("Cloth Image URL:", clothImageUrl); // Cloth image is already a URL

      // Send images for Try-On processing
      const tryOnResponse = await axios.post(
        "http://localhost:5050/api/v1/tryon",
        {
          target_file_url: userImageUrl, // User image
          source_file_url: clothImageUrl, // Cloth image
        }
      );

      console.log("Try-On API Response:", tryOnResponse.data);

      if (tryOnResponse.data?.result_url) {
        setResultUrl(tryOnResponse.data.result_url);
      } else {
        setErrorMessage("Try-On result not available.");
      }
    } catch (error) {
      console.error("Try-On failed:", error);
      setErrorMessage("An error occurred during the Try-On process.");
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="try-on-container">
      <h2>Try On Clothes</h2>

      <div className="image-upload">
        <label>Upload Your Image:</label>
        <input type="file" accept="image/*" onChange={handleFileChange} />
      </div>

      {clothImageUrl && (
        <div>
          <h4>Selected Cloth:</h4>
          <img src={clothImageUrl} alt="Selected Cloth" width="200" />
        </div>
      )}

      <button onClick={handleTryOn} disabled={loading}>
        {loading ? "Processing..." : "Try On"}
      </button>

      {resultUrl && (
        <div>
          <h4>Try-On Result:</h4>
          <img src={resultUrl} alt="Try-On Result" width="200" />
        </div>
      )}

      {errorMessage && <p className="error">{errorMessage}</p>}
    </div>
  );
};

export default TryOnPage;
